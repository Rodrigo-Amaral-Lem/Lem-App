cmake_minimum_required(VERSION 3.25)
project(Lem VERSION 0.1.57 LANGUAGES CXX)
# Normal Qt Creator builds use the selected shared Qt kit.
# The standalone packaging script explicitly enables the static-only option.
option(LEM_WINDOWS_STANDALONE "Link Windows Qt/plugins/runtime into one Lem.exe" OFF)
if(LEM_WINDOWS_STANDALONE AND MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2 Network Concurrent)
# App-level response limits do not replace Qt's internal HTTP parser fix.
# Patched vendor kits may retain 6.12.0: their backport must be verified separately.
if(Qt6_VERSION VERSION_LESS "6.12.1")
 message(WARNING "Qt ${Qt6_VERSION}: verify/apply upstream CVE-2026-116337 Qt Network patches, or use Qt 6.12.1+ when available. Lem does not patch the installed Qt SDK. See docs/security-review-0.1.33.md.")
endif()
qt_standard_project_setup(REQUIRES 6.12)

lem_deploy_android_openssl(Lem)

set_source_files_properties(qml/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
set_source_files_properties(qml/Measurements.js PROPERTIES QT_RESOURCE_ALIAS Measurements.js)
# QML's URI creates a Lem directory; keep it away from the Linux executable.
# Also expose the base path to Qt's QML tooling/import scanner.
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml")
qt_add_qml_module(Lem
 URI Lem VERSION 1.0
 OUTPUT_DIRECTORY "${QT_QML_OUTPUT_DIRECTORY}/Lem"
 QML_FILES qml/SignatureDialog.qml qml/Main.qml qml/Measurements.js qml/PhotoCropDialog.qml
)

if(WIN32)
 enable_language(RC)
 if(MINGW)
  # windres forwards inherited -I paths incorrectly when the workspace contains spaces.
  # This icon-only resource needs no Qt include paths; compile it in its asset directory.
  set(lem_icon_object "${CMAKE_CURRENT_BINARY_DIR}/lem-icon.o")
  add_custom_command(OUTPUT "${lem_icon_object}"
   COMMAND "${CMAKE_RC_COMPILER}" -O coff -i lem.rc -o "${lem_icon_object}"
   WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assets"
   DEPENDS assets/lem.rc assets/lem.ico
   VERBATIM)
  set_source_files_properties("${lem_icon_object}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
  target_sources(Lem PRIVATE "${lem_icon_object}")
 else()
  target_sources(Lem PRIVATE assets/lem.rc)
  target_include_directories(Lem PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
 endif()
endif()
qt_add_resources(Lem schema PREFIX / BASE "${CMAKE_CURRENT_SOURCE_DIR}/.." FILES ../shared/checklist-v1.json)

lem_configure_windows_standalone(Lem)
if(MSVC)
 target_compile_options(Lem PRIVATE /utf-8 /W4)
 target_compile_definitions(Lem PRIVATE NOMINMAX)
elseif(MINGW)
 target_compile_options(Lem PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()

endif()

install(TARGETS Lem
 BUNDLE DESTINATION .
 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
)

endif()

 install(SCRIPT ${deploy_script})
