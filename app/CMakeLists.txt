cmake_minimum_required(VERSION 3.25)
project(Lem VERSION 0.6.0 LANGUAGES CXX)
set(report_limits_file "${CMAKE_CURRENT_SOURCE_DIR}/../shared/report-limits.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${report_limits_file}")
file(READ "${report_limits_file}" report_limits)
foreach(limit IN ITEMS payload_utf8_bytes answer_characters general_note_characters)
 string(JSON limit_value GET "${report_limits}" "${limit}")
 string(TOUPPER "${limit}" limit_name)
 add_compile_definitions("LEM_${limit_name}=${limit_value}")
endforeach()
# Normal Qt Creator builds use the selected shared Qt kit.
# The standalone packaging script explicitly enables the static-only option.
option(LEM_WINDOWS_STANDALONE "Link Windows Qt/plugins/runtime into one Lem.exe" OFF)
if(LEM_WINDOWS_STANDALONE AND MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2 Network Concurrent Svg)
# App-level response limits do not replace Qt's internal HTTP parser fix.
# Patched vendor kits may retain 6.12.0: their backport must be verified separately.
if(Qt6_VERSION VERSION_LESS "6.12.1")
 message(WARNING "Qt ${Qt6_VERSION}: verify/apply upstream CVE-2026-116337 Qt Network patches, or use Qt 6.12.1+ when available. Lem does not patch the installed Qt SDK. See docs/build-and-release.md.")
endif()
qt_standard_project_setup(REQUIRES 6.12)
# The standalone Qt/cache lives under app/build. Scanning generated Qt sources
# once per test target makes every reconfigure needlessly slow and finds foreign QML.
list(APPEND QT_QML_IMPORT_SCANNER_EXTRA_ARGS -exclude build)

 enable_language(RC)
 if(MINGW)
  # windres forwards inherited -I paths incorrectly when the workspace contains spaces.
  # This icon-only resource needs no Qt include paths; compile it in its asset directory.
  set(lem_icon_object "${CMAKE_CURRENT_BINARY_DIR}/lem-icon.o")
  add_custom_command(OUTPUT "${lem_icon_object}"
   COMMAND "${CMAKE_RC_COMPILER}" -O coff -i lem.rc -o "${lem_icon_object}"
   WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assets"
   DEPENDS assets/lem.rc assets/lem.ico
   VERBATIM)
  set_source_files_properties("${lem_icon_object}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
  target_sources(Lem PRIVATE "${lem_icon_object}")

  target_sources(Lem PRIVATE assets/lem.rc)
  target_include_directories(Lem PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
 endif()

  set_property(TARGET ${lem_target} PROPERTY QT_QML_MODULE_NO_IMPORT_SCAN TRUE)

  target_sources(${lem_target} PRIVATE src/SessionStore.cpp src/SessionStore.h)
  if(WIN32)
   target_link_libraries(${lem_target} PRIVATE crypt32)
  elseif(APPLE)
   target_link_libraries(${lem_target} PRIVATE "-framework Security" "-framework CoreFoundation")
  endif()
 endif()

  target_link_libraries(${lem_target} PRIVATE Qt6::Qml)
